-- DATABASE MANAGEMENT - PROJECT 1
-- User Management System
--
-- SQL statements used by the application.
--
-- NOTE: Statements containing ? are parameterized queries
-- executed by the Node.js backend. They are not intended
-- to be run directly in phpMyAdmin as written.



CREATE DATABASE IF NOT EXISTS web_app;

USE web_app;

CREATE TABLE Users (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(255) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    salary DECIMAL(12,2),
    age INT,
    registerday DATE NOT NULL,
    signintime DATETIME NULL
);

-- REQUIREMENT 1: REGISTER A NEW USER

INSERT INTO Users
(
    username,
    password,
    firstname,
    lastname,
    salary,
    age,
    registerday,
    signintime
)
VALUES (?, ?, ?, ?, ?, ?, CURDATE(), NULL);


-- REQUIREMENT 2: USER SIGN-IN

SELECT username, password
FROM Users
WHERE username = ?;

-- After bcrypt verifies the password, update signintime.

UPDATE Users
SET signintime = NOW()
WHERE username = ?;


-- REQUIREMENT 3: SEARCH BY FIRST AND/OR LAST NAME

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE 1 = 1
  AND firstname LIKE ?
  AND lastname LIKE ?
ORDER BY lastname, firstname;


-- REQUIREMENT 4: SEARCH BY USER ID (USERNAME)

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE username = ?;


-- REQUIREMENT 5: SEARCH BY SALARY RANGE

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE salary BETWEEN ? AND ?
ORDER BY salary ASC;


-- REQUIREMENT 6: SEARCH BY AGE RANGE

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE age BETWEEN ? AND ?
ORDER BY age ASC;


-- REQUIREMENT 7: REGISTERED AFTER JOHN

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE registerday > (
    SELECT registerday
    FROM Users
    WHERE username = ?
)
ORDER BY registerday ASC;


-- REQUIREMENT 8: USERS WHO NEVER SIGNED IN

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE signintime IS NULL
ORDER BY username;


-- REQUIREMENT 9: REGISTERED ON SAME DAY AS JOHN

-- The backend supplies 'john' as the username parameter.

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE registerday = (
    SELECT registerday
    FROM Users
    WHERE username = ?
)
ORDER BY username;


-- REQUIREMENT 10: REGISTERED TODAY

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
WHERE registerday = CURDATE()
ORDER BY username;


-- ADDITIONAL FEATURE: VIEW ALL USERS

SELECT username, firstname, lastname,
       salary, age, registerday, signintime
FROM Users
ORDER BY username;



-- OPTIONAL TEST DATA: CHANGE REGISTRATION DATES

-- Original test statement from the project.
-- Sets Bob's registration date one day after John's.

UPDATE Users
SET registerday = DATE_ADD(
    (SELECT john_date FROM (
        SELECT registerday AS john_date
        FROM Users
        WHERE username = 'john'
    ) AS dates),
    INTERVAL 1 DAY
)
WHERE username = 'bob';

-- 1. CREATE THE NAMES TABLE

CREATE TABLE IF NOT EXISTS names (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100),
    date_added DATE
);


-- 2. INSERT SAMPLE RECORDS

INSERT INTO names (name, date_added)
VALUES
('John', CURDATE()),
('Alice', CURDATE()),
('Bob', CURDATE() - INTERVAL 1 DAY);


-- 3. VIEW ALL RECORDS

SELECT * FROM names;


-- 4. SEARCH FOR A SPECIFIC NAME

SELECT * FROM names
WHERE name = 'Alice';


-- 5. FIND NAMES ADDED TODAY

SELECT * FROM names
WHERE date_added = CURDATE();


-- 6. UPDATE AN EXISTING RECORD

UPDATE names
SET name = 'Robert'
WHERE name = 'Bob';


-- 7. DELETE A RECORD

DELETE FROM names
WHERE name = 'Robert';


-- 8. COUNT THE TOTAL RECORDS

SELECT COUNT(*) AS total_names
FROM names;
